cmake_minimum_required(VERSION 3.23)
project(agent03_vision_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(SQLite3 REQUIRED)
get_filename_component(AOI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(aoi_a4_vision STATIC "${AOI_ROOT}/libs/vision/src/pipeline.cpp")
target_include_directories(aoi_a4_vision PUBLIC "${AOI_ROOT}/libs/vision/include" "${AOI_ROOT}/libs/data/include")
target_link_libraries(aoi_a4_vision PUBLIC ${OpenCV_LIBS} OpenSSL::Crypto)
add_library(aoi_a4_models STATIC "${AOI_ROOT}/libs/models/src/model_registry.cpp")
target_include_directories(aoi_a4_models PUBLIC "${AOI_ROOT}/libs/models/include")
target_link_libraries(aoi_a4_models PUBLIC OpenSSL::Crypto)
add_library(aoi_a6_wal STATIC "${AOI_ROOT}/libs/data/src/wal_store.cpp")
target_include_directories(aoi_a6_wal PUBLIC "${AOI_ROOT}/libs/data/include")
target_link_libraries(aoi_a6_wal PUBLIC OpenSSL::Crypto SQLite::SQLite3)
add_library(aoi_a4_replay STATIC "${AOI_ROOT}/libs/replay/src/lab.cpp")
target_include_directories(aoi_a4_replay PUBLIC "${AOI_ROOT}/libs/replay/include")
target_link_libraries(aoi_a4_replay PUBLIC aoi_a4_vision aoi_a6_wal ${OpenCV_LIBS})
add_executable(aoi_agent03_vision_test "${AOI_ROOT}/tests/unit/vision/native_vision_lab_test.cpp")
target_link_libraries(aoi_agent03_vision_test PRIVATE aoi_a4_replay aoi_a4_models)
include(CTest)
add_test(NAME aoi_agent03_vision COMMAND aoi_agent03_vision_test "${AOI_ROOT}/db/migrations/0002_agent03_simonly_draft.sql")
