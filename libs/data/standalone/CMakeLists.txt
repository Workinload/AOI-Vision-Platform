cmake_minimum_required(VERSION 3.23)
project(aoi_agent03_native_data LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX)
  message(FATAL_ERROR "Agent03 POSIX fsync native prototype: Windows port NOT_VERIFIED")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(SQLite3 REQUIRED)
get_filename_component(AOI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(aoi_agent03_wal STATIC "${AOI_ROOT}/libs/data/src/wal_store.cpp")
target_include_directories(aoi_agent03_wal PUBLIC "${AOI_ROOT}/libs/data/include")
target_link_libraries(aoi_agent03_wal PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
add_library(aoi_agent03_assets STATIC "${AOI_ROOT}/libs/assets/src/lossless_store.cpp")
target_include_directories(aoi_agent03_assets PUBLIC "${AOI_ROOT}/libs/assets/include")
target_link_libraries(aoi_agent03_assets PUBLIC aoi_agent03_wal)
add_library(aoi_agent03_statistics STATIC "${AOI_ROOT}/libs/statistics/src/report.cpp")
target_include_directories(aoi_agent03_statistics PUBLIC "${AOI_ROOT}/libs/statistics/include")
target_link_libraries(aoi_agent03_statistics PUBLIC SQLite::SQLite3)
add_executable(agent03_native_store_test "${AOI_ROOT}/tests/unit/storage/native_wal_asset_test.cpp")
target_link_libraries(agent03_native_store_test PRIVATE aoi_agent03_assets aoi_agent03_statistics)
include(CTest)
add_test(NAME agent03_native_wal_asset COMMAND agent03_native_store_test
         "${AOI_ROOT}/db/migrations/0002_agent03_simonly_draft.sql")
