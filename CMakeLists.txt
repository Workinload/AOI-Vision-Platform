cmake_minimum_required(VERSION 3.25)
project(AOIVisionPlatform VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AOI_SIM_ONLY "Build a simulator with no physical PLC/camera driver" ON)
option(AOI_BUILD_GUI "Build Qt Quick UI (Qt6.11.2)" ON)
option(AOI_PROBE_OPENCV5 "If OpenCV5 SDK available, build/test real core+imgproc probe" ON)
if(NOT AOI_SIM_ONLY)
    message(FATAL_ERROR "M00 only supports AOI_SIM_ONLY=ON. Physical PLC/camera access is deliberately absent.")
endif()
add_compile_definitions(AOI_SIM_ONLY=1)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Network Sql)
if(AOI_BUILD_GUI)
    find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Gui Qml Quick QuickControls2)
endif()
qt_standard_project_setup(REQUIRES 6.11)
add_library(aoi_domain INTERFACE)
target_include_directories(aoi_domain INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/libs/domain/include")
target_compile_features(aoi_domain INTERFACE cxx_std_20)
qt_add_executable(aoi-core apps/aoi-core/main.cpp)
target_link_libraries(aoi-core PRIVATE aoi_domain Qt6::Core Qt6::Network Qt6::Sql)
qt_add_executable(vplc simulators/vplc/main.cpp)
target_link_libraries(vplc PRIVATE Qt6::Core Qt6::Network)
if(AOI_BUILD_GUI)
    qt_add_executable(aoi-ui apps/aoi-ui/main.cpp apps/aoi-ui/StatusBridge.cpp apps/aoi-ui/StatusBridge.hpp)
    qt_add_qml_module(aoi-ui URI AOI.UI VERSION 1.0 QML_FILES resources/qml/Main.qml)
    target_link_libraries(aoi-ui PRIVATE aoi_domain Qt6::Core Qt6::Network Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(AOI_PROBE_OPENCV5)
    find_package(OpenCV 5.0.0 EXACT QUIET COMPONENTS core imgproc)
    if(OpenCV_FOUND)
        add_executable(opencv5_link_probe tests/unit/opencv5_link_probe.cpp)
        target_include_directories(opencv5_link_probe PRIVATE ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(opencv5_link_probe PRIVATE ${OpenCV_LIBS})
        message(STATUS "AOI M00 OpenCV5 probe enabled: ${OpenCV_VERSION}")
        if(BUILD_TESTING)
            add_test(NAME opencv5_link_probe COMMAND opencv5_link_probe)
        endif()
    else()
        message(WARNING "OpenCV5 Linux SDK not found: image processing remains NOT_VERIFIED; set CMAKE_PREFIX_PATH to exact 5.0.0 SDK")
    endif()
endif()
if(BUILD_TESTING)
    add_executable(domain_contract_test tests/unit/domain_contract_test.cpp)
    target_link_libraries(domain_contract_test PRIVATE aoi_domain)
    add_test(NAME domain_contract_test COMMAND domain_contract_test)
    add_test(NAME patch_protocol_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/patch/test_zip_protocol.py")
    add_test(NAME negative_guard_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/negative_guards.py" --bin-dir "$<TARGET_FILE_DIR:aoi-core>")
    add_test(NAME source_contracts_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/unit/source_contracts_test.py" "${CMAKE_CURRENT_SOURCE_DIR}")
    if(AOI_BUILD_GUI)
        add_test(NAME five_process_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/five_process_smoke.py" --bin-dir "$<TARGET_FILE_DIR:aoi-core>")
        set_tests_properties(five_process_smoke PROPERTIES TIMEOUT 35 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software;QT_QUICK_BACKEND=software;QT_PLUGIN_PATH=$ENV{QT_PLUGIN_PATH};QML2_IMPORT_PATH=$ENV{QML2_IMPORT_PATH}")
    endif()
endif()
message(STATUS "AOI M00: SIM_ONLY=ON, GUI=${AOI_BUILD_GUI}, Qt=${Qt6_VERSION} (FINS/Camera/Windows unimplemented)")
